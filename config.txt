
Programmi
	
	progs = {
		Programma: [ Giorni, Orari, Durate ],
		...
	}

	esempio:
	
		progs = {
			'a': [['every', 1], [hm(9,30), hm(11,30)], [sm(2,4), sm(1,6)]],
			'b': [['even'], [hm(9,30)], [sm(3,5), sm(4,2)]],
			'c': [['dow', 1, 1, 0, 1, 1, 0, 1], [hm(10,30)], [sm(9,1), sm(1,5)]],
		}

Giorni (di irrigazione per programma) (howOften)

	['every', n] # ogni n giorni
	['odd']      # giorni dispari
	['even']     # giorni pari
	['o/e', n]   # giorni pari o dispari a seconda che n sia 0 o 1 
	['dow', [L, Ma, Me, G, V, S, D]] # L, Ma, Me, G, V, S, D a seconda che siano 0 o 1 
			
Orari (di partenza per programma) (timeStart)

	[ hm(ore,minuti), ... ]
	
Durate (di irrigazione per programma e stazione) (runTime)

	[ sm(stazione, minuti), ... ] # con stazione fra 1 ed 8 e minuti tra 1 e 60
